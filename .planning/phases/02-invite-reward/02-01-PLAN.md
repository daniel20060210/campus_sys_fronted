  description: string
  createdAt: number

  id: number
  name: string
  description: string
  startTime: number
  endTime: number
  status: 0 | 1 | 2          // 0=未开始, 1=进行中, 2=已结束
  totalInviters: number
  totalInvitees: number
  rewardBudget: number
  rewardUsed: number
  tiers: RewardTier[]
  createdAt: number
  updatedAt: number
}

interface InviteeRecord {
  id: number
  inviterId: number
  inviterNickname: string
  inviterAvatar: string
  inviteeId: number
  inviteeNickname: string
  inviteeAvatar: string
  schoolName: string
  registeredAt: number
  orderCount: number
  totalOrderAmount: number
  rewardReceived: number
}
